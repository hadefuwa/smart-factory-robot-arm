you can ssh onto the pi using pi@rpi but do not make changes to the pi, make changes here on my windows machine and then push to github and pull on the pi like this..

cd C:\Users\HamedA\Documents\sf2
git status
git add .
git commit -m "bridge issues"
git push origin main
ssh pi@rpi "cd ~/sf2 && git pull"
ssh pi@rpi "cd ~/sf2 && git pull && sudo systemctl restart smart-factory.service robotarmv3-pi.service"

git tag -a v5.0.2 -m "Release v5.0.2"; git push origin v5.0.2

live status:
ssh pi@rpi "sudo systemctl is-active smart-factory.service robotarmv3-pi.service"
ssh pi@rpi "curl -sk https://localhost:8080/api/robot-arm/status | python3 -m json.tool"
ssh pi@rpi "sudo journalctl -u smart-factory.service -u robotarmv3-pi.service -f"
